// Second reference run for pkfind: automatic height threshold and the sign rule.
// A = x = 0..24 step 0.1; B = eleven triangular peaks (half width 0.5) at x = 2, 4, ..., 22 with heights
// 1, .6, .3, .24, .22, .2, .19, .15, .12, .1, .05; C = B + 0.5; D = B - 2; E = B - 0.5 (mixed sign); F = -B.

col(A) = data(0, 24, 0.1);
col(B) = 1.0*(1-abs(col(A)-2)/0.5)*(abs(col(A)-2)<0.5) + 0.6*(1-abs(col(A)-4)/0.5)*(abs(col(A)-4)<0.5) + 0.3*(1-abs(col(A)-6)/0.5)*(abs(col(A)-6)<0.5) + 0.24*(1-abs(col(A)-8)/0.5)*(abs(col(A)-8)<0.5) + 0.22*(1-abs(col(A)-10)/0.5)*(abs(col(A)-10)<0.5) + 0.2*(1-abs(col(A)-12)/0.5)*(abs(col(A)-12)<0.5) + 0.19*(1-abs(col(A)-14)/0.5)*(abs(col(A)-14)<0.5) + 0.15*(1-abs(col(A)-16)/0.5)*(abs(col(A)-16)<0.5) + 0.12*(1-abs(col(A)-18)/0.5)*(abs(col(A)-18)<0.5) + 0.1*(1-abs(col(A)-20)/0.5)*(abs(col(A)-20)<0.5) + 0.05*(1-abs(col(A)-22)/0.5)*(abs(col(A)-22)<0.5);
col(C) = col(B) + 0.5;
col(D) = col(B) - 2;
col(E) = col(B) - 0.5;
col(F) = -col(B);
